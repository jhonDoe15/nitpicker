function double(value: number): number {
  return value * 2;
}

const holder = {
  factor: 3,
  times(value: number): number {
    return value * this.factor;
  },
};

export function direct(): number {
  return double(2);
}

export function method(): number {
  return holder.times(2);
}

export function probe(): unknown {
  return [direct(), method()];
}
