export function strict(flag: boolean): string {
  return flag ? 'yes' : 'no';
}

export function negated(flag: boolean): string {
  return flag ? 'yes' : 'no';
}

export function nullable(flag: boolean | undefined): string {
  return flag === false ? 'off' : 'on-or-unknown';
}

export function probe(): unknown {
  return [strict(true), negated(false), nullable(undefined)];
}
